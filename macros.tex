%% Macro to print versicles
\newcommand{\versiculus}[4]{
	\twocoltext{
		\vv #1 \\ \rr #2
	}{
		\vv #3 \\ \rr #4
	}
}

%% Little Chapter
\newcommand{\capitulum}[3]{%
	%% #1 : biblical reference
	%% #2 : little chapter text, with \psstar and \pscross for the flexes and mediants
	%% #3 : translation
	\needspace{3\baselineskip} % To avoid orphaning the title
	\hspace*{0.04\textwidth}% See also the padding of the rubrics macro
	\textcolor{gregoriocolor}{Capitule}%
	\hfill%
	\textcolor{gregoriocolor}{#1}%
	\hspace*{0.04\textwidth}%
	\twocoltext{\rdrop #2}{\rdrop #3}
	\vspace{2mmplus 1.5mmminus 1.5mm}
}

%% Little Chapter
\newcommand{\lbrev}[3]{%
	%% #1 : biblical reference
	%% #2 : lectio brevis text, with \psstar and \pscross for the flexes and mediants
	%% #3 : translation
	\needspace{3\baselineskip} % To avoid orphaning the title
	\hspace*{0.04\textwidth}% See also the padding of the rubrics macro
	\textcolor{gregoriocolor}{Lecture Brève}%
	\hfill%
	\textcolor{gregoriocolor}{#1}%
	\hspace*{0.04\textwidth}%
	\twocoltext{\rdrop #2}{\rdrop #3}
	\vspace{2mmplus 1.5mmminus 1.5mm}
}

\newcommand{\oratio}[2]{%
	%% #1 : oration text, with \psstar and \pscross for the flexes and mediants
	\needspace{3\baselineskip} % To avoid orphaning the title
	{\centering\textcolor{gregoriocolor}{Oraison}\par}
	\twocoltext{\rdrop #1}{\rdrop #2}
}
