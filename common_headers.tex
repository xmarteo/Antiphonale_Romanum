}

%% Given a name header (from Gregorio header capture),
%% this section labels the piece with the piece label and label name,
%% and indexes the piece accordint to its office-part and mode
\begin{luacode*}
function delete_newline ( s )
   s = string.gsub ( s, 'newline', '')
   s = string.gsub ( s, 'hspace', '')
   s = string.gsub ( s, '4mm', '')
   s = string.gsub ( s, 'protect', '')
   s = string.gsub ( s, 'hbox', '')
   s = string.gsub ( s, [[\]], '')
   s = string.gsub ( s, "{}", '')
   tex.sprint ( s )
end
\end{luacode*}
\makeatletter
\newcommand{\namecapture}[1]{%
  %% this prevents page breaks between the phantom section and its label, and the actual score.
  \needspace{4\baselineskip}%
  \protected@edef\@currentlabelname{\directlua{delete_newline(\luastring{#1})}}%
  \phantomsection%
  \label{\labelbuffer}%
  %% indexation
  \ifdefstrequal{H}{\indexnamebuffer}{%
    \index[\indexnamebuffer]{{#1}@\indexedmode & #1}%
  }{}%
  \ifdefstrequal{A}{\indexnamebuffer}{%
    \index[\indexnamebuffer]{{#1}@\indexedmode & #1}%
  }{}%
  \ifdefstrequal{R}{\indexnamebuffer}{%
    \index[\indexnamebuffer]{#1}%
  }{}%
}
\makeatother

\newcommand{\gscore}[3]{

  %% #3 : the translation of the piece.

  \translation{#3}
  \resetinitialstyle

%% Macro to print an itemized list on two columns (psalm, canticle, etc) from a tex file
\newcommand{\psalmverseindent}{10pt}
\newcommand{\twocolitemized}[2]{
	\vspace{2mm plus .5mm minus .5mm}
	\begin{paracol}{2}
	\begin{itemize}[
		label=\null, 
		leftmargin=0pt, 
		itemindent=\psalmverseindent, 
		labelsep=0pt, 
		labelwidth=0pt, 
		rightmargin=0pt, 
		parsep=0pt, 
		itemsep=0pt plus 1pt minus 0pt,
		topsep=-2mm,
		]
	\input{#1}
	\end{itemize}
	\switchcolumn
	\begin{itemize}[
		label=\null, 
		leftmargin=0pt, 
		itemindent=\psalmverseindent, 
		labelsep=0pt, 
		labelwidth=0pt, 
		rightmargin=0pt, 
		parsep=0pt, 
		itemsep=0pt plus 1pt minus 0pt,
		topsep=-2mm,
		]
	\input{#2}
	%% We add the French GP if and only if this is not TriumPuerorum
	\ifstrequal{#2}{psalmi_fr/cantici_lettrine/TriumPuerorum}%
		{}%
		{\item Gloire au Père, au F\frflex{i}ls, et au Saint-Esprit,\psstar{} pour les siècles des si\frflex{è}cles, Amen.}
	\end{itemize}
	\end{paracol}
	\vspace{3mm plus 1mm minus 1mm}
}

\newcommand{\psalmus}[1]{
	%% #1: psalm number
	\needspace{4\baselineskip}
	\phantomsection
	\label{Psalm#1_\nextpsalmmode}
	\index[P]{#1 (mode \nextpsalmmode)}
	\vspace{2mm plus .5mm minus .5mm}
	{\centering\textbf{Psaume #1}\par}
	\twocolitemized{psalmi_la/psalmi_lettrine/#1-\nextpsalmmode}{psalmi_fr/psalmi_lettrine/#1}
	\resetlettrinestyle
}

\newcommand{\canticum}[3]{
	%% #1: canticle code (for file naming purposes and indexing purposes}
	%% #2: canticle title, to be printed
	%% #3: biblical reference
	\needspace{4\baselineskip}
	\vspace{2mm plus .5mm minus .5mm}
	\phantomsection
	\label{Canticum#1_\nextpsalmmode}
	\index[C]{#2 (mode \nextpsalmmode)}
	\ifstrequal{#3}{}{%
		{\centering\textbf{#2}\par}% Cas rare qui gère des cantiques non-scripturaires. Ex: Symbole d'Athanase
	}{%
		\hspace*{0.04\textwidth}% See also the padding of the rubrics macro
		\textbf{#2}%
		\hfill%
		\textcolor{gregoriocolor}{#3}%
		\hspace*{0.04\textwidth}%
	}
	\twocolitemized{psalmi_la/cantici_lettrine/#1-\nextpsalmmode}{psalmi_fr/cantici_lettrine/#1}
	\resetlettrinestyle
}
